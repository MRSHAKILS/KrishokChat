\begin{adjustbox}{width=\textwidth}
\begin{tikzpicture}[
    node distance=0.8cm and 1.2cm,
    font=\sffamily\footnotesize,
    >=Stealth,
    thick,
    box/.style={rectangle, rounded corners=3pt, draw=black!70, fill=white, text width=2.4cm, align=center, minimum height=0.9cm, drop shadow={opacity=0.15, shadow xshift=1pt, shadow yshift=-1pt}},
    inputbox/.style={box, fill=blue!8, draw=blue!60, font=\sffamily\bfseries\footnotesize},

    nlubox/.style={box, fill=orange!10, draw=orange!70},
    visionbox/.style={box, fill=teal!10, draw=teal!70},
    ragbox/.style={box, fill=green!10, draw=green!70},
    verifybox/.style={box, fill=purple!10, draw=purple!70},
    uibox/.style={box, fill=gray!12, draw=gray!70, font=\sffamily\bfseries\footnotesize},
    decision/.style={diamond, draw=black!70, fill=yellow!30!orange!30, aspect=1.8, inner sep=1pt, text width=1.8cm, align=center, font=\sffamily\scriptsize},
    arrow/.style={->, draw=black!80, line width=0.8pt},
    dasharrow/.style={->, dashed, draw=black!70, line width=0.8pt},
    groupbox/.style={draw=black!40, dashed, rounded corners=4pt, inner sep=6pt, fill=black!2}
]

    % Inputs
    \node[inputbox] (input_text) {Dialectal Bengali / Voice Query};
    \node[inputbox, below=0.6cm of input_text] (input_img) {Smartphone Leaf Photo};

    % Tier 0: Safety Screening
    \node[safetybox, right=1.2cm of input_text] (safety_gate) {\textbf{Tier 0: Safety Gate}\\ \scriptsize Precheck Regex + Fast Rule ($0.32\,$ms)};
    \node[safetybox, above=0.7cm of safety_gate, fill=red!20, draw=red!80] (emergency) {\textbf{16123 Escalation}\\ \scriptsize Krishi Call Center / Crisis};

    % Multimodal Vision Router
    \node[visionbox, right=1.2cm of input_img] (vision_router) {\textbf{On-Device Vision}\\ \scriptsize 6 ONNX Classifiers ($26\text{--}265\,$ms WASM)};

    % Tier 1: Cost-Gated Disambiguation
    \node[nlubox, right=1.2cm of safety_gate] (nlu_gate) {\textbf{Semantic NLU}\\ \scriptsize Cost-Gated Slot Parser (\texttt{Gemini-2.5-Flash-Lite} / Regex)};
    \node[decision, right=1.0cm of nlu_gate] (crop_check) {Crop Slot\\Present?};
    \node[nlubox, above=0.7cm of crop_check, fill=yellow!25!orange!20, draw=orange!70] (clarify_turn) {\textbf{Interactive Clarify}\\ \scriptsize Quick-Reply Crop Chips (Saves $88\%$ Tokens)};

    % Cross-modal conflict check
    \node[decision, right=1.0cm of vision_router] (conflict_check) {$c_{\text{img}} == c_{\text{txt}}$\\Aligned?};
    \node[visionbox, below=0.6cm of conflict_check, fill=red!10, draw=red!60] (conflict_badge) {\textbf{Mismatch Warning}\\ \scriptsize Flag Contradiction};

    % Tier 2 / Tier 3 Retrieval & Resolution Ladder
    \node[ragbox, right=1.4cm of crop_check] (retrieval) {\textbf{Hybrid RAG (T3)}\\ \scriptsize BM25 ($26.8\,$ms) + Dense BGE-M3 (2,946 Docs)};
    \node[ragbox, below=0.6cm of retrieval] (generator) {\textbf{Gemma-4 4-bit LoRA}\\ \scriptsize Grounded Generation via SSE Streaming};

    % Tier 4 Verification & Safe Action Output
    \node[verifybox, right=1.2cm of generator] (verifier) {\textbf{Relational Verifier}\\ \scriptsize Dosage \& Interval Gate ($4.82\,$ms Fail-Closed)};

    % Final Outputs
    \node[uibox, right=1.2cm of verifier] (action_card) {\textbf{Dual-Audience UI}\\ \scriptsize Safe Action Card + ``Why this advice?'' Trace};

    % Background groupings
    \begin{scope}[on background layer]
        \node[groupbox, fit=(safety_gate) (emergency), label={[anchor=north, font=\sffamily\bfseries\scriptsize, text=red!70!black]north:Deterministic Safety Rail}] (g_safety) {};
        \node[groupbox, fit=(nlu_gate) (crop_check) (clarify_turn), label={[anchor=north, font=\sffamily\bfseries\scriptsize, text=orange!70!black]north:Cost-Gated Semantic Disambiguation}] (g_nlu) {};
        \node[groupbox, fit=(retrieval) (generator) (verifier), label={[anchor=north, font=\sffamily\bfseries\scriptsize, text=green!60!black]north:Evidence-Grounded Advisory Engine}] (g_rag) {};
    \end{scope}

    % Connecting Arrows
    \draw[arrow] (input_text) -- (safety_gate);
    \draw[arrow] (safety_gate) -- node[left, font=\scriptsize, text=red!80] {Unsafe} (emergency);
    \draw[arrow] (safety_gate) -- node[above, font=\scriptsize] {Safe} (nlu_gate);

    \draw[arrow] (input_img) -- (vision_router);
    \draw[arrow] (vision_router) -- (conflict_check);
    \draw[arrow] (conflict_check) -- node[right, font=\scriptsize, text=red!80] {No} (conflict_badge);
    \draw[arrow] (conflict_check) -| node[near end, above, font=\scriptsize] {Yes} (retrieval);

    \draw[arrow] (nlu_gate) -- (crop_check);
    \draw[arrow] (crop_check) -- node[left, font=\scriptsize, text=orange!80] {No} (clarify_turn);
    \draw[arrow] (crop_check) -- node[above, font=\scriptsize] {Yes} (retrieval);
    \draw[dasharrow] (clarify_turn) -| node[near start, above, font=\scriptsize] {User Crop Reply} (input_text);

    \draw[arrow] (retrieval) -- (generator);
    \draw[arrow] (generator) -- (verifier);
    \draw[arrow] (verifier) -- node[above, font=\scriptsize] {Verified} (action_card);

\end{tikzpicture}
