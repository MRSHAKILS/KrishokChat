    inputbox/.style={box, fill=blue!8, draw=blue!60, font=\sffamily\scriptsize},
    visionbox/.style={box, fill=teal!10, draw=teal!70, font=\sffamily\scriptsize},
    conflictbox/.style={box, fill=red!10, draw=red!70, font=\sffamily\scriptsize},
    actionbox/.style={box, fill=green!10, draw=green!70, font=\sffamily\scriptsize},
    decision/.style={diamond, draw=black!70, fill=yellow!30!orange!30, aspect=1.8, inner sep=1pt, text width=1.6cm, align=center, font=\sffamily\scriptsize},

    % Image Path
    \node[inputbox] (img_in) {\textbf{Multimodal User Input:}\\ Photo (Potato Leaf) + Text (\textit{``amar begun gache poka''})};

    \node[visionbox, below=0.5cm of img_in] (crop_clf) {\textbf{Stage 1: Crop Identification}\\ 6-Class ONNX Classifier ($26\text{--}45\,$ms WASM)\\ Output: $c_{\text{img}} = \textbf{Potato}$};

    \node[visionbox, below=0.5cm of crop_clf] (disease_clf) {\textbf{Stage 2: Disease Diagnostic}\\ Crop-Specific ONNX (\texttt{potato\_disease.onnx})\\ Output: $d_{\text{img}} = \textbf{Late Blight}$ (Conf: $0.94$)};

    \node[decision, below=0.6cm of disease_clf] (check_conflict) {Cross-Modal\\Alignment\\($c_{\text{img}} == c_{\text{txt}}$)?};

    \node[conflictbox, right=0.8cm of check_conflict] (mismatch_modal) {\textbf{Contradiction Disambiguation (S3)}\\ \scriptsize Flag $c_{\text{img}} \neq c_{\text{txt}}$ ($85.0\%$ detection rate)\\ Prompt: \textit{``Image indicates potato leaf. Do you want advice for potato or brinjal?''}};

    \node[actionbox, below=0.7cm of check_conflict] (action_card) {\textbf{Safe Action Card Generation (S2)}\\ \scriptsize Approved Active: \textbf{Mancozeb 75 WP} ($2\,$g/L)\\ Safe Envelope: $7\text{--}10$ day spray interval\\ Pre-Harvest Interval (PHI): $14$ days};

    % Arrows
    \draw[arrow] (img_in) -- (crop_clf);
    \draw[arrow] (crop_clf) -- (disease_clf);
    \draw[arrow] (disease_clf) -- (check_conflict);
    \draw[arrow] (check_conflict) -- node[above, font=\scriptsize, text=red!70!black] {No ($c_{\text{img}} \neq c_{\text{txt}}$)} (mismatch_modal);
    \draw[arrow] (check_conflict) -- node[left, font=\scriptsize, text=green!60!black] {Yes (Aligned)} (action_card);
    \draw[dasharrow] (mismatch_modal) |- (action_card);
