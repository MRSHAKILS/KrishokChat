    \node[userbox] (q1) {\textbf{Underspecified Symptom Query:}\\ \textit{``patay dag parche, ki dimu?''}\\ \scriptsize (``Spots on leaves, what to apply?'')};

    \node[routerbox, below=0.6cm of q1] (slot_extract) {\textbf{Semantic NLU Slot Extraction}\\ \scriptsize $\text{kind}=\text{treatment}, \text{part}=\text{leaf}$\\ $\text{symptom}=\text{spot}, \mathbf{crop}=\mathbf{\emptyset}$};

    \node[decision, below=0.6cm of slot_extract] (check_crop) {Crop Slot\\Missing?};

    \node[clarifybox, right=0.8cm of check_crop] (clarify) {\textbf{Clarification Intercept (T-Clarify)}\\ \scriptsize Bypasses Blind RAG ($\mathbf{88\%}$ Token Savings)\\ \textit{``Which crop has this leaf problem?''}\\ Quick-Reply Chips: [Rice] [Potato] [Tomato]};

    \node[userbox, above=0.7cm of clarify] (user_reply) {\textbf{Farmer 1-Tap Crop Selection}\\ User taps: \textbf{[Potato]} (\textit{Aloo})};

    \node[routerbox, above=0.7cm of user_reply] (rewrite) {\textbf{Conversational Query Rewriter}\\ Synthesizes canonical query:\\ \textit{``What is the remedy for potato leaf spots?''}};

    \node[ragbox, left=0.8cm of rewrite] (rag) {\textbf{Grounded Agricultural RAG}\\ Precise BARI Late Blight retrieval $\rightarrow$ Verified active chemical dosage};

    % Arrows
    \draw[arrow] (q1) -- (slot_extract);
    \draw[arrow] (slot_extract) -- (check_crop);
    \draw[arrow] (check_crop) -- node[above, font=\scriptsize, text=red!70!black] {Yes ($crop = \emptyset$)} (clarify);
    \draw[arrow] (check_crop) -- node[left, font=\scriptsize, text=green!60!black] {No} (rag);
    \draw[dasharrow] (clarify) -- (user_reply);
    \draw[arrow] (user_reply) -- (rewrite);
    \draw[arrow] (rewrite) -- (rag);
